&#x20;   void parseGGA(const std::string& line) {

&#x20;       // \$GNGGA,123519,4807.038,N,01131.000,E,1,08,0.9,545.4,M,46.9,M,,\*47

&#x20;       std::vector\<std::string> tokens = split(line, ',');

&#x20;       if (tokens.size() < 15) return;

&#x20;      &#x20;

&#x20;       try {

&#x20;           // 时间

&#x20;           std::string time\_str = tokens\[1];

&#x20;           int hour = std::stoi(time\_str.substr(0, 2));

&#x20;           int minute = std::stoi(time\_str.substr(2, 2));

&#x20;           int second = std::stoi(time\_str.substr(4, 2));

&#x20;          &#x20;

&#x20;           // 纬度

&#x20;           double lat\_deg = std::stod(tokens\[2].substr(0, 2));

&#x20;           double lat\_min = std::stod(tokens\[2].substr(2));

&#x20;           double latitude = lat\_deg + lat\_min / 60.0;

&#x20;           if (tokens\[3] == "S") latitude = -latitude;

&#x20;          &#x20;

&#x20;           // 经度

&#x20;           double lon\_deg = std::stod(tokens\[4].substr(0, 3));

&#x20;           double lon\_min = std::stod(tokens\[4].substr(3));

&#x20;           double longitude = lon\_deg + lon\_min / 60.0;

&#x20;           if (tokens\[5] == "W") longitude = -longitude;

&#x20;          &#x20;

&#x20;           // 海拔

&#x20;           double altitude = std::stod(tokens\[9]);

&#x20;          &#x20;

&#x20;           // 定位状态

&#x20;           int fix\_quality = std::stoi(tokens\[6]);

&#x20;           sensor\_msgs::NavSatStatus status;

&#x20;           switch (fix\_quality) {

&#x20;               case 0: status.status = sensor\_msgs::NavSatStatus::STATUS\_NO\_FIX; break;

&#x20;               case 1: status.status = sensor\_msgs::NavSatStatus::STATUS\_FIX; break;

&#x20;               case 2: status.status = sensor\_msgs::NavSatStatus::STATUS\_SBAS\_FIX; break;

&#x20;               case 4: status.status = sensor\_msgs::NavSatStatus::STATUS\_GBAS\_FIX; break;

&#x20;               case 5: status.status = sensor\_msgs::NavSatStatus::STATUS\_GBAS\_FIX; break;

&#x20;               default: status.status = sensor\_msgs::NavSatStatus::STATUS\_NO\_FIX; break;

&#x20;           }

&#x20;          &#x20;

&#x20;           // 发布定位数据

&#x20;           sensor\_msgs::NavSatFix fix\_msg;

&#x20;           fix\_msg.header.stamp = ros::Time(hour \* 3600 + minute \* 60 + second);

&#x20;           fix\_msg.header.frame\_id = "gps\_link";

&#x20;           fix\_msg.status = status;

&#x20;           fix\_msg.latitude = latitude;

&#x20;           fix\_msg.longitude = longitude;

&#x20;           fix\_msg.altitude = altitude;

&#x20;          &#x20;

&#x20;           // 设置协方差（根据定位质量估算）

&#x20;           double covariance\[9] = {0.0};

&#x20;           switch (status.status) {

&#x20;               case sensor\_msgs::NavSatStatus::STATUS\_NO\_FIX:

&#x20;                   std::fill(covariance, covariance + 9, 1e10);

&#x20;                   break;

&#x20;               case sensor\_msgs::NavSatStatus::STATUS\_FIX:

&#x20;                   covariance\[0] = covariance\[4] = 100.0;  // 100m精度

&#x20;                   covariance\[8] = 1000.0;

&#x20;                   break;

&#x20;               case sensor\_msgs::NavSatStatus::STATUS\_SBAS\_FIX:

&#x20;                   covariance\[0] = covariance\[4] = 5.0;    // 5m精度

&#x20;                   covariance\[8] = 100.0;

&#x20;                   break;

&#x20;               case sensor\_msgs::NavSatStatus::STATUS\_GBAS\_FIX:

&#x20;                   covariance\[0] = covariance\[4] = 0.1;    // 0.1m精度

&#x20;                   covariance\[8] = 10.0;

&#x20;                   break;

&#x20;           }

&#x20;          &#x20;

&#x20;           std::copy(covariance, covariance + 9, fix\_msg.position\_covariance.begin());

&#x20;           fix\_msg.position\_covariance\_type = sensor\_msgs::NavSatFix::COVARIANCE\_TYPE\_KNOWN;

&#x20;          &#x20;

&#x20;           gps\_pub\_.publish(fix\_msg);

&#x20;          &#x20;

&#x20;       } catch (std::exception& e) {

&#x20;           ROS\_WARN("Failed to parse GGA message: %s", e.what());

&#x20;       }

&#x20;   }

&#x20;   void parseRMC(const std::string& line) {

&#x20;       // \$GNRMC,123519,A,4807.038,N,01131.000,E,022.4,084.4,230394,003.1,W\*6A

&#x20;       std::vector\<std::string> tokens = split(line, ',');

&#x20;       if (tokens.size() < 13) return;

&#x20;      &#x20;

&#x20;       try {

&#x20;           // 时间

&#x20;           std::string time\_str = tokens\[1];

&#x20;           int hour = std::stoi(time\_str.substr(0, 2));

&#x20;           int minute = std::stoi(time\_str.substr(2, 2));

&#x20;           int second = std::stoi(time\_str.substr(4, 2));

&#x20;          &#x20;

&#x20;           // 日期

&#x20;           std::string date\_str = tokens\[9];

&#x20;           int day = std::stoi(date\_str.substr(0, 2));

&#x20;           int month = std::stoi(date\_str.substr(2, 2));

&#x20;           int year = std::stoi(date\_str.substr(4, 2)) + 2000;

&#x20;          &#x20;

&#x20;           // 速度（节）

&#x20;           double speed\_knots = std::stod(tokens\[7]);

&#x20;           double speed\_ms = speed\_knots \* 0.514444;  // 转换为m/s

&#x20;          &#x20;

&#x20;           // 航向角

&#x20;           double course = std::stod(tokens\[8]);

&#x20;          &#x20;

&#x20;           // 磁偏角

&#x20;           double magnetic\_variation = std::stod(tokens\[11]);

&#x20;          &#x20;

&#x20;           ROS\_DEBUG("RMC: Time %02d:%02d:%02d, Date %02d/%02d/%04d, Speed %.2f m/s, Course %.1f",

&#x20;                     hour, minute, second, day, month, year, speed\_ms, course);

&#x20;          &#x20;

&#x20;       } catch (std::exception& e) {

&#x20;           ROS\_WARN("Failed to parse RMC message: %s", e.what());

&#x20;       }

&#x20;   }

&#x20;   std::vector\<std::string> split(const std::string& s, char delimiter) {

&#x20;       std::vector\<std::string> tokens;

&#x20;       std::string token;

&#x20;       std::istringstream tokenStream(s);

&#x20;       while (std::getline(tokenStream, token, delimiter)) {

&#x20;           tokens.push\_back(token);

&#x20;       }

&#x20;       return tokens;

&#x20;   }

## 四、代码实现要点说明

### 4.1 硬件通信协议实现要点

**USB 通信（Orbbec 相机和 RTK）**：



* 使用 RK3588 SDK 提供的`uart_driver.h`接口

* 通过`open()`、`write()`、`read()`函数进行串口通信

* 支持波特率配置：9600、19200、38400、57600、115200

* 设备映射：相机默认`/dev/ttyUSB0`，RTK 默认`/dev/ttyUSB1`

**以太网通信（Hesai 激光雷达）**：



* 使用 RK3588 SDK 的 ETH 驱动

* 通过 UDP 协议通信（点云端口 2368，PTC 端口 9347）

* 主机需配置静态 IP：192.168.1.100

* 雷达 IP：192.168.1.200（需与主机同网段）

### 4.2 ROS1 接口规范实现要点

**标准消息格式实现**：



1. **点云消息（PointCloud2）**：

* 用于 Orbbec 相机深度数据和 Hesai 激光雷达

* 包含 x、y、z 坐标和强度信息

* 时间戳使用 ROS 系统时间

* 坐标系：相机使用`camera_depth_frame`，激光雷达使用`hesai_lidar_frame`

1. **IMU 消息（Imu）**：

* 用于 Hesai 激光雷达的惯性测量单元

* 包含角速度、线加速度和协方差信息

* 发布频率：10Hz

1. **GPS 定位消息（NavSatFix）**：

* 用于 UM982 RTK 定位数据

* 包含经纬度、海拔和定位状态

* 定位状态映射：


  * 0: 无定位

  * 1: GPS 定位

  * 2: DGPS 定位

  * 4: RTK 固定解

  * 5: RTK 浮点解

### 4.3 工程编译与运行要点

**编译环境要求**：



* Ubuntu 20.04 LTS

* ROS Noetic

* RK3588 交叉编译工具链

* 依赖库：libusb-1.0-dev、libyaml-cpp-dev、libopencv-dev

**运行方式**：

2. 仅传感器：`roslaunch launch sensor_only.launch`

3. API 调试：`roslaunch launch api_debug.launch`
